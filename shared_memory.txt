#include <stdio.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <string.h>

int main() {
    int id;
    char *ptr;

    id = shmget(IPC_PRIVATE, 1024, IPC_CREAT | 0666);
    if (id == -1) {
        perror("shmget");
        return 1;
    }

    ptr = (char *)shmat(id, NULL, 0);
    if (ptr == (char *)-1) {
        perror("shmat");
        shmctl(id, IPC_RMID, NULL);
        return 1;
    }

    strcpy(ptr, "Hello from shared memory");

    printf("Data written: %s\n", ptr);
    printf("Data read: %s\n", ptr);

    shmdt(ptr);
    shmctl(id, IPC_RMID, NULL);

    return 0;
}
